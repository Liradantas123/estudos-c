 bool encerrar = false;

 while (true) {

  if (detectar_sd()) {

   led_piscando();
   inicializar_spi();
   inicializar_sd();
   disponibilizar_usb(); 

  while (true) {

   transferencia_e_edicao_de_dados();
  
    if (ejetado_sd()) {
     led_on();
     encerrar = true;
     break;
      }
   
    if (remocao_inesperada_sd()) {
     led_error();
     encerrar = true;
     break;
     }

   }

 }

else {
led_on();
}

if (encerrar) {
break;
}

}

cancelar_detecao_sd();
final();
}
